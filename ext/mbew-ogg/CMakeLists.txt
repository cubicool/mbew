project(mbew-ogg)

add_subdirectory("ogg" EXCLUDE_FROM_ALL)

foreach(VAR OGG_HEADERS OGG_SOURCES)
    get_directory_property(VAR_FILES DIRECTORY "ogg" DEFINITION ${VAR})

    foreach(VF ${VAR_FILES})
        # OGG_HEADERS now includes a build-tree-generated header (config_types.h) as an absolute
        # path - only relative entries need the "ogg/" prefix.
        if(IS_ABSOLUTE "${VF}")
            list(APPEND MBEW_OGG_FILES "${VF}")
        else()
            list(APPEND MBEW_OGG_FILES "ogg/${VF}")
        endif()
    endforeach()
endforeach()

add_library(mbew-ogg STATIC ${MBEW_OGG_FILES})

target_include_directories(mbew-ogg PUBLIC
    "${PROJECT_SOURCE_DIR}/ogg/include"
    # config_types.h is generated by ogg's own configure_file() into ITS build dir, not the
    # source tree - needed alongside the source include dir above for <ogg/config_types.h>.
    "${CMAKE_CURRENT_BINARY_DIR}/ogg/include"
)
